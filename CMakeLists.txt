target_compile_features(steammaster_dsp PUBLIC cxx_std_17)
if (MSVC)

# Optional JUCE VST3 test bench. Keeps the DSP-only CI path independent of JUCE.
option(STEAMMASTER_BUILD_VST3 "Build SteamMaster EZ VST3" OFF)
if(STEAMMASTER_BUILD_VST3)
  include(FetchContent)
  FetchContent_Declare(juce GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(juce)
  juce_add_plugin(SteamMasterEZ COMPANY_NAME "ThomC" BUNDLE_ID "com.thomc.steammaster.ez" PLUGIN_MANUFACTURER_CODE ThmC PLUGIN_CODE SmE1 FORMATS VST3 PRODUCT_NAME "SteamMaster EZ" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
  target_sources(SteamMasterEZ PRIVATE plugin/PluginProcessor.cpp plugin/PluginProcessor.h plugin/PluginEditor.cpp plugin/PluginEditor.h)
  target_link_libraries(SteamMasterEZ PRIVATE steammaster_dsp juce::juce_audio_utils juce::juce_audio_processors juce::juce_gui_extra juce::juce_dsp)
  target_compile_definitions(SteamMasterEZ PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
  juce_generate_juce_header(SteamMasterEZ)
  if(BUILD_TESTING)
    add_executable(steammaster_plugin_host_smoke tests/TestPluginHost.cpp)
    target_link_libraries(steammaster_plugin_host_smoke PRIVATE SteamMasterEZ steammaster_dsp juce::juce_audio_utils)
    target_include_directories(steammaster_plugin_host_smoke PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/plugin)
    add_test(NAME SteamMasterPluginHostSmoke COMMAND steammaster_plugin_host_smoke)
  endif()
endif()
